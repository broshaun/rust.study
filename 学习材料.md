import { create } from 'zustand'


type AuthState = {
  loading: boolean
  cacheReady: boolean
  error: Error | null
  initCache: () => Promise<{uid: any, db: any}>
}

export const useAuthStore = create<AuthState>((set, get) => ({
  loading: false,
  cacheReady: false,
  error: null,

  initCache: async () => {
    if (get().cacheReady) {
      const uid = userId.get()
      return { uid, db: getUserDB(uid) }
    }
    set({ loading: true, error: null })
    try {
      const uid = userId.get()
      const db = getUserDB(uid)
      await loginCache2.fetch()
      set({ cacheReady: true })
      return { uid, db }
    } catch (err) {
      set({ error: err as Error })
      throw err // 依然抛出，上层可自定义弹窗提示
    } finally {
      set({ loading: false })
    }
  },
}))
